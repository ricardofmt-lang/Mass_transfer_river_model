'Model parameters


Sub Channel()

'Main activities:
      '1-reads parameters
      '2-computes concentrations
      '3-outputs results
    
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    
' A0 - Variables Declaration
    Dim i As Integer
    Dim ctrl As Controls
    Dim printCount As Integer
    
    Dim gr As grid
    Dim flow As Flow_properties
    Dim Coef As Equation_Coef
    Dim atm As Atmosphere
    
    Dim pr As PropertiesList
 
 'Initializes the model
 
 'Reads control parameters
    ctrl = Setcontrols()
 'Reads grid parameters
    gr = CreateGrid()
 'Reads flow parameters
    flow = CreateFlow(gr, ctrl)
 'Allocates memory for transport coeficients
    Coef = CreateCoef(gr.NC)
    
'Initializes properties options and values
    
    pr.Generic = CreateProperty("Generic", gr, ctrl)
    pr.Temperature = CreateProperty("Temperature", gr, ctrl)
    pr.DO = CreateProperty("DO", gr, ctrl)
    pr.BOD = CreateProperty("BOD", gr, ctrl)
    pr.CO2 = CreateProperty("CO2", gr, ctrl)
    
    
' Reads Atmosphere Parameters Must apperar after  property creation
    atm = CreateAtmosphere(pr)
    
    If pr.DO.Active And Not pr.Temperature.Active Then
      MsgBox ("DO active requires Temperature to be active too")
     End
    End If

    If pr.BOD.Active And Not pr.DO.Active Then
      MsgBox ("BOD active requires DO to be active too")
     Stop
    End If
    
    If pr.CO2.Active And Not (pr.BOD.Active Or pr.DO.Active) Then
      MsgBox ("CO2 Active require both BOD and DO to be active too")
     Stop
    End If
    
    
 'Prints Initial values
        If pr.Generic.Active Then
           Call PrintValues(pr.Generic, gr, ctrl)
        End If
        
        If pr.Temperature.Active Then
            Call PrintValues(pr.Temperature, gr, ctrl)
        End If
        
        If pr.DO.Active Then
            Call PrintValues(pr.DO, gr, ctrl)
        End If
        
        If pr.BOD.Active Then
            Call PrintValues(pr.BOD, gr, ctrl)
        End If

        If pr.CO2.Active Then
            Call PrintValues(pr.CO2, gr, ctrl)
        End If
' Updates printing control
    Call UpdatePrinting(ctrl)
        
 'Cycles to compute Properties values from start time up to end time
    Do While ctrl.T < ctrl.SimulationDuration
        
 'Sets new time and zeros coefficients
        ctrl.T = ctrl.T + ctrl.dt
        ctrl.timedays = ctrl.T / 86400
        Call ZerosCoeffs(Coef, gr)
        
'Applies discharges in two halfs. If discharges are applied before transport the maximum value appears in cell after the discharge in the output
'if discharges are applied only after transport a peack is created in the discharge cell, that increases with the time step. Applying discharges in two half steps minimises the problems
        If pr.Generic.Active And pr.Generic.Boundary.PropDischarges.DischargesActive Then
           Call ApplyDischarges(pr.Generic, gr, ctrl.dt)
        End If
        
        If pr.Temperature.Active Then
        
          If pr.Temperature.Boundary.PropDischarges.DischargesActive Then
            Call ApplyDischarges(pr.Temperature, gr, ctrl.dt)
          End If
          
           If pr.Temperature.Boundary.CyclicBoundary Then
              If flow.velocity > 0 Then
                 pr.Temperature.Boundary.LeftValue = pr.Temperature.NewValue(gr.NC)
              Else
                 pr.Temperature.Boundary.RightValue = pr.Temperature.NewValue(1)
              End If
            End If 'CyclicBoundary
           
           End If 'pr.Temperature.Active
        
        If pr.DO.Active And pr.DO.Boundary.PropDischarges.DischargesActive Then
            Call ApplyDischarges(pr.DO, gr, ctrl.dt)
        End If
        
        If pr.BOD.Active And pr.BOD.Boundary.PropDischarges.DischargesActive Then
            Call ApplyDischarges(pr.BOD, gr, ctrl.dt)
        End If
        
        If pr.CO2.Active And pr.CO2.Boundary.PropDischarges.DischargesActive Then
            Call ApplyDischarges(pr.CO2, gr, ctrl.dt)
        End If
        
' Performs Transport
        Call CoefTransport(flow, gr.NC, ctrl, Coef)
         
        If pr.Generic.Active Then
           Call Transport(pr.Generic, Coef, gr.NC, ctrl)
        End If
        
        If pr.Temperature.Active Then
            Call Transport(pr.Temperature, Coef, gr.NC, ctrl)
        End If
        
        If pr.DO.Active Then
            Call Transport(pr.DO, Coef, gr.NC, ctrl)
        End If
        
        If pr.BOD.Active Then
            Call Transport(pr.BOD, Coef, gr.NC, ctrl)
        End If
        
         If pr.CO2.Active Then
            Call Transport(pr.CO2, Coef, gr.NC, ctrl)
        End If
       
        
' Computes Free surface fluxes

        If pr.Temperature.Active And pr.Temperature.Boundary.FreeSurfaceFlux Then
             Call ExpFreeSurfaceHeatFluxes(pr.Temperature, gr, ctrl, atm)
        End If
         
        If pr.DO.Active And pr.DO.Boundary.FreeSurfaceFlux Then
             Call ExpFreeSurfaceGasFluxes(pr.Temperature, pr.DO, gr, ctrl, flow, atm)
        End If
        
        If pr.CO2.Active And pr.CO2.Boundary.FreeSurfaceFlux Then
             Call ExpFreeSurfaceGasFluxes(pr.Temperature, pr.CO2, gr, ctrl, flow, atm)
        End If
        
  'Updates OD/BOD concentrations by Organic Matter Oxidation
        
        If pr.BOD.Active Then
            Call ExpSinksBOD(pr, gr, ctrl.dt) ' always computed explicitly
        End If
        
        If (pr.Generic.Active And pr.Generic.DecayOrGrowth) Then
             If (ctrl.TimeDiscretisation = "exp") Then
                Call ExpSinksGeneric(pr.Generic, gr.NC, ctrl.dt)
             Else
                Call ImpTypeSinksGeneric(pr.Generic, gr.NC, ctrl)
             End If
        End If 'pr.BOD.Active
        
 'Prints if it is time
If ctrl.T >= ctrl.printtime Then
           
        If pr.Generic.Active Then
           Call PrintValues(pr.Generic, gr, ctrl)
        End If
        
        If pr.Temperature.Active Then
            Call PrintValues(pr.Temperature, gr, ctrl)
           
        '   Sheets("Temperature").Select
        '    ActiveSheet.Cells(ctrl.PrintLine, 313) = Solar_Radiation(ctrl.timedays, atm)

        End If
        
        If pr.DO.Active Then
            Call PrintValues(pr.DO, gr, ctrl)
        End If
        
        If pr.BOD.Active Then
            Call PrintValues(pr.BOD, gr, ctrl)
        End If
           
        If pr.CO2.Active Then
            Call PrintValues(pr.CO2, gr, ctrl)
        End If
           
        Call UpdatePrinting(ctrl)
End If 'ctrl.T
                
    Loop
        
'After reaching  endtime, creates graphs
    Call CreateGraphs(pr, ctrl, gr.NC)
        
    Sheets("Main").Select
        
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    
    MsgBox ("The Application Ended")
        
End Sub

Sub CoefTransport(flow As Flow_properties, NC As Integer, ctrl As Controls, Coef As Equation_Coef)
    
    Dim i As Integer


    If ctrl.Diffusion Then
    
    ' Diffusion: no Flux at the boundary
        Coef.A(1) = 0
        Coef.B(1) = 0
        Coef.e(1) = -flow.DiffusionNr
        Coef.f(1) = flow.DiffusionNr
        Coef.g(1) = 0
        
        Coef.A(NC) = 0
        Coef.B(NC) = flow.DiffusionNr
        Coef.e(NC) = -flow.DiffusionNr
        Coef.f(NC) = 0
        Coef.g(NC) = 0
 
' inner points
    For i = 2 To NC - 1
        Coef.A(i) = 0
        Coef.B(i) = flow.DiffusionNr
        Coef.e(i) = -2 * flow.DiffusionNr
        Coef.f(i) = flow.DiffusionNr
        Coef.g(i) = 0
      Next
      
    End If 'ctrl.Diffusion
    
    If ctrl.Advection Then
    
      'Computes  coefficients as if it was Upwind. to correct the Quick melhod later
         
      If ctrl.AdvType = "upwind" Then ' Advection Upwind
      
          If flow.velocity > 0 Then
            For i = 1 To NC
                Coef.B(i) = Coef.B(i) + flow.CourantNr
                Coef.e(i) = Coef.e(i) - flow.CourantNr
                Coef.f(i) = Coef.f(i)
            Next
          Else
            For i = 1 To NC
                Coef.B(i) = Coef.B(i)
                Coef.e(i) = Coef.e(i) + flow.CourantNr
                Coef.f(i) = Coef.f(i) - flow.CourantNr
            Next
       End If
        
      
      ElseIf ctrl.AdvType = "QUICK" Then ' Advection Upwind
       
       If ctrl.QUICK_UP Then
           If flow.velocity > 0 Then
             For i = 1 To NC
                Coef.b_up(i) = Coef.B(i) + flow.CourantNr
                Coef.e_up(i) = Coef.e(i) - flow.CourantNr
                Coef.f_up(i) = Coef.f(i)
             Next
           Else
             For i = 1 To NC
                Coef.b_up(i) = Coef.B(i)
                Coef.e_up(i) = Coef.e(i) + flow.CourantNr
                Coef.f_up(i) = Coef.f(i) - flow.CourantNr
             Next
           End If 'Flow.velocity
        End If 'QUICK_UP
      
          If flow.velocity > 0 Then
          
              For i = 2 To NC - 1
      'contribution of left face concentration
                Coef.A(i) = Coef.A(i) - 1 / 8 * flow.CourantNr
                Coef.B(i) = Coef.B(i) + 6 / 8 * flow.CourantNr
                Coef.e(i) = Coef.e(i) + 3 / 8 * flow.CourantNr
     'contribution of right face concentration
                Coef.B(i) = Coef.B(i) + 1 / 8 * flow.CourantNr
                Coef.e(i) = Coef.e(i) - 6 / 8 * flow.CourantNr
                Coef.f(i) = Coef.f(i) - 3 / 8 * flow.CourantNr
              Next
                Coef.B(1) = Coef.B(1) + 9 / 8 * flow.CourantNr
                Coef.e(1) = Coef.e(1) - 6 / 8 * flow.CourantNr
                Coef.f(1) = Coef.f(1) - 3 / 8 * flow.CourantNr
                
                Coef.A(NC) = Coef.A(NC) - 1 / 8 * flow.CourantNr
                Coef.B(NC) = Coef.B(NC) + 6 / 8 * flow.CourantNr
                Coef.e(NC) = Coef.e(NC) - 5 / 8 * flow.CourantNr
                Coef.f(NC) = 0
            
          Else ' flow.velocity<0
          
              For i = 2 To NC - 1
       'contribution of left face concentration
                Coef.B(i) = Coef.B(i) + 3 / 8 * flow.CourantNr
                Coef.e(i) = Coef.e(i) + 6 / 8 * flow.CourantNr
                Coef.f(i) = Coef.f(i) - 1 / 8 * flow.CourantNr
       'contribution of right face concentration
                Coef.e(i) = Coef.e(i) - 3 / 8 * flow.CourantNr
                Coef.f(i) = Coef.f(i) - 6 / 8 * flow.CourantNr
                Coef.g(i) = Coef.g(i) + 1 / 8 * flow.CourantNr
              Next
            
                Coef.B(1) = 0
                Coef.e(1) = Coef.e(1) + 5 / 8 * flow.CourantNr
                Coef.f(1) = Coef.f(1) - 6 / 8 * flow.CourantNr
                Coef.g(1) = Coef.g(1) + 1 / 8 * flow.CourantNr
                
                Coef.B(NC) = Coef.B(NC) + 3 / 8 * flow.CourantNr
                Coef.e(NC) = Coef.e(NC) + 6 / 8 * flow.CourantNr
                Coef.f(NC) = Coef.f(NC) - 9 / 8 * flow.CourantNr
           
          End If 'Flow.velocity

        
      ElseIf ctrl.AdvType = "central" Then
      
            For i = 2 To NC - 1
               Coef.B(i) = Coef.B(i) + flow.CourantNr / 2
               Coef.f(i) = Coef.f(i) - flow.CourantNr / 2
            Next
            
            If flow.velocity > 0 Then   ' Even with central differences upwind has to be used at the boundaries
                Coef.B(1) = Coef.B(1) + flow.CourantNr
                Coef.e(1) = Coef.e(1) - flow.CourantNr
                Coef.B(NC) = Coef.B(NC) + flow.CourantNr
                Coef.e(NC) = Coef.e(NC) - flow.CourantNr
            Else
                Coef.e(1) = Coef.e(1) + flow.CourantNr
                Coef.f(1) = Coef.f(1) - flow.CourantNr
                Coef.e(NC) = Coef.e(NC) + flow.CourantNr
                Coef.f(NC) = Coef.f(NC) - flow.CourantNr
            End If
      Else
         MsgBox "Advection Type not defined:" & ctrl.AdvType
      End If
       
    End If
    
End Sub
Sub Transport(Prop As Property, Coef As Equation_Coef, NC As Integer, ctrl As Controls)

    If ctrl.AdvType = "QUICK" Then
       If ctrl.TimeDiscretisation = "exp" Then
            Call ExpQUICKTransport(Prop, Coef, NC, ctrl)
       ElseIf ctrl.TimeDiscretisation = "imp" Then
            Call ImpQUICKTransport(Prop, Coef, NC, ctrl)
       Else
            Call SemiImpQUICKTransport(Prop, Coef, NC, ctrl)
       End If
    Else
       If ctrl.TimeDiscretisation = "exp" Then
            Call ExpTransport(Prop, Coef, NC)
       ElseIf ctrl.TimeDiscretisation = "imp" Then
            Call ImpTransport(Prop, Coef, NC)
       Else
            Call semiImpTransport(Prop, Coef, NC)
       End If
    End If


End Sub
Sub ExpTransport(Prop As Property, Coef As Equation_Coef, NC As Integer)
    
    Dim i As Integer


'The last computated values are stored in "Old" and coefficient "CenterCoef" is updated for explicit
    For i = 1 To NC
        Prop.Old(i) = Prop.NewValue(i)
        Coef.LeftCoef(i) = Coef.B(i)
        Coef.CenterCoef(i) = (1 + Coef.e(i))
        Coef.RigthCoef(i) = Coef.f(i)
    Next i

'Left Boundary
    Prop.NewValue(1) = Coef.LeftCoef(1) * Prop.Boundary.LeftValue + Coef.CenterCoef(1) * Prop.Old(1) + Coef.RigthCoef(1) * Prop.Old(2)

'Inner Points
    For i = 2 To NC - 1
        Prop.NewValue(i) = Coef.LeftCoef(i) * Prop.Old(i - 1) + Coef.CenterCoef(i) * Prop.Old(i) + Coef.RigthCoef(i) * Prop.Old(i + 1)
    Next i
'Right Boundary
        Prop.NewValue(NC) = Coef.LeftCoef(NC) * Prop.Old(NC - 1) + Coef.CenterCoef(NC) * Prop.Old(NC) + Coef.RigthCoef(NC) * Prop.Boundary.RightValue
         
End Sub
Sub ExpQUICKTransport(Prop As Property, Coef As Equation_Coef, NC As Integer, ctrl As Controls)
        
    Dim i As Integer
    Dim A, B, C As Double
 
'The last computated values are stored in "Old" and coefficient "CenterCoef" is updated for explicit
    For i = 1 To NC
    
        Prop.Old(i) = Prop.NewValue(i)
        
        Coef.L_LeftCoef(i) = Coef.A(i)
        Coef.LeftCoef(i) = Coef.B(i)
        Coef.CenterCoef(i) = (1 + Coef.e(i))
        Coef.RigthCoef(i) = Coef.f(i)
        Coef.R_RigthCoef(i) = Coef.g(i)
        
    Next i
    
If ctrl.QUICK_UP Then

    For i = 2 To NC - 1

          C = WorksheetFunction.Max(Abs(Prop.Old(i + 1) - Prop.Old(i - 1)), ctrl.QUICK_UP_Ratio * (1 + Prop.minimum)) ' It uses (1 + Prop.minimum) to work also whenProp.minimum is zero.
          A = WorksheetFunction.Max(Abs(Prop.Old(i) - Prop.Old(i - 1)), ctrl.QUICK_UP_Ratio * (1 + Prop.minimum))     ' If Prop.minimum is zero the values of A, B and C could very small and the test below
          B = WorksheetFunction.Max(Abs(Prop.Old(i + 1) - Prop.Old(i)), ctrl.QUICK_UP_Ratio * (1 + Prop.minimum))     ' could be verified in areas where property values are negligeable

      If (Abs(A - C) > ctrl.QUICK_UP_Ratio * A Or Abs(B - C) > ctrl.QUICK_UP_Ratio * B) Then ' uses upwind
         Coef.L_LeftCoef(i) = 0
         Coef.LeftCoef(i) = Coef.b_up(i)
         Coef.CenterCoef(i) = 1 + Coef.e_up(i)
         Coef.RigthCoef(i) = Coef.f_up(i)
         Coef.R_RigthCoef(i) = 0
      End If
      
    Next i
    
End If ' ctrl.QUICK_UP


'Left Boundary
    Prop.NewValue(1) = Coef.LeftCoef(1) * Prop.Boundary.LeftValue + Coef.CenterCoef(1) * Prop.Old(1) + Coef.RigthCoef(1) * Prop.Old(2) + Coef.R_RigthCoef(1) * Prop.Old(3)
    Prop.NewValue(2) = Coef.L_LeftCoef(2) * Prop.Boundary.LeftValue + Coef.LeftCoef(2) * Prop.Old(1) + Coef.CenterCoef(2) * Prop.Old(2) + Coef.RigthCoef(1) * Prop.Old(3) + Coef.R_RigthCoef(1) * Prop.Old(4)

'Inner Points
    For i = 3 To NC - 2
        Prop.NewValue(i) = Coef.L_LeftCoef(i) * Prop.Old(i - 2) + Coef.LeftCoef(i) * Prop.Old(i - 1) + Coef.CenterCoef(i) * Prop.Old(i) + Coef.RigthCoef(i) * Prop.Old(i + 1) + Coef.R_RigthCoef(i) * Prop.Old(i + 2)
    Next i
'Right Boundary
        Prop.NewValue(NC - 1) = Coef.L_LeftCoef(NC - 1) * Prop.Old(NC - 3) + Coef.LeftCoef(NC - 1) * Prop.Old(NC - 2) + Coef.CenterCoef(NC - 1) * Prop.Old(NC - 1) + Coef.RigthCoef(NC - 1) * Prop.Old(NC - 1) + Coef.R_RigthCoef(NC - 1) * Prop.Boundary.RightValue
        Prop.NewValue(NC) = Coef.L_LeftCoef(NC) * Prop.Old(NC - 2) + Coef.LeftCoef(NC) * Prop.Old(NC - 1) + Coef.CenterCoef(NC) * Prop.Old(NC) + Coef.RigthCoef(NC) * Prop.Boundary.RightValue
         
End Sub
Sub ImpTransport(Prop As Property, Coef As Equation_Coef, NC As Integer)
    
    Dim i, iaux As Integer
    Dim f As Double
   
'The last computated values are stored in "Old"
    For i = 1 To NC
        Prop.Old(i) = Prop.NewValue(i)
        Coef.TiCoef(i) = Prop.Old(i)
        Coef.LeftCoef(i) = -Coef.B(i)
        Coef.CenterCoef(i) = 1 - Coef.e(i)
        Coef.RigthCoef(i) = -Coef.f(i)
    Next i

'Left Boundary
    Coef.TiCoef(1) = Coef.TiCoef(1) - Coef.LeftCoef(1) * Prop.Boundary.LeftValue
    Coef.TiCoef(NC) = Coef.TiCoef(NC) - Coef.RigthCoef(NC) * Prop.Boundary.RightValue
    

'Inner Points
    For i = 2 To NC
       f = Coef.LeftCoef(i) / Coef.CenterCoef(i - 1)
       Coef.LeftCoef(i) = Coef.LeftCoef(i) - f * Coef.CenterCoef(i - 1)
       Coef.CenterCoef(i) = Coef.CenterCoef(i) - f * Coef.RigthCoef(i - 1)
       Coef.TiCoef(i) = Coef.TiCoef(i) - f * Coef.TiCoef(i - 1)
    Next i
'Right Boundary

    Prop.NewValue(NC) = Coef.TiCoef(NC) / Coef.CenterCoef(NC)
    
    For iaux = 1 To NC - 1
    i = NC - iaux
    Prop.NewValue(i) = (Coef.TiCoef(i) - Coef.RigthCoef(i) * Prop.NewValue(i + 1)) / Coef.CenterCoef(i)
    Next iaux
        
End Sub
Sub semiImpTransport(Prop As Property, Coef As Equation_Coef, NC As Integer)
    
    Dim i, iaux As Integer
    Dim f As Double
   
'The last computated values are stored in "Old"
    For i = 1 To NC
    
        Prop.Old(i) = Prop.NewValue(i)
        
        Coef.TiCoef(i) = Prop.Old(i)
        Coef.LeftCoef(i) = -Coef.B(i) / 2
        Coef.CenterCoef(i) = 1 - Coef.e(i) / 2
        Coef.RigthCoef(i) = -Coef.f(i) / 2
    Next i

'Left Boundary
    Coef.TiCoef(1) = Coef.TiCoef(1) - Coef.LeftCoef(1) * Prop.Boundary.LeftValue + 0.5 * (Coef.B(1) * Prop.Boundary.LeftValue + Coef.e(1) * Prop.Old(1) + Coef.f(1) * Prop.Old(2))
    Coef.TiCoef(NC) = Coef.TiCoef(NC) - Coef.RigthCoef(NC) * Prop.Boundary.RightValue + 0.5 * (Coef.B(NC) * Prop.Old(NC - 1) + Coef.e(NC) * Prop.Old(NC) + Coef.f(NC) * Prop.Boundary.RightValue)
    
    For i = 2 To NC - 1
        Coef.TiCoef(i) = Coef.TiCoef(i) + 0.5 * (Coef.B(i) * Prop.Old(i - 1) + Coef.e(i) * Prop.Old(i) + Coef.f(i) * Prop.Old(i + 1))
    Next i

'Inner Points
    For i = 2 To NC
       f = Coef.LeftCoef(i) / Coef.CenterCoef(i - 1)
       Coef.LeftCoef(i) = Coef.LeftCoef(i) - f * Coef.CenterCoef(i - 1)
       Coef.CenterCoef(i) = Coef.CenterCoef(i) - f * Coef.RigthCoef(i - 1)
       Coef.TiCoef(i) = Coef.TiCoef(i) - f * Coef.TiCoef(i - 1)
    Next i
'Right Boundary

    Prop.NewValue(NC) = Coef.TiCoef(NC) / Coef.CenterCoef(NC)
    
    For iaux = 1 To NC - 1
    i = NC - iaux
    Prop.NewValue(i) = (Coef.TiCoef(i) - Coef.RigthCoef(i) * Prop.NewValue(i + 1)) / Coef.CenterCoef(i)
    Next iaux
        
End Sub

Sub ApplyDischarges(Prop As Property, gr As grid, dt As Double)
    
    Dim i, Ncell As Integer
    Dim VolumeRate, SpecificValue As Double
    Dim aux As Double
    

    For i = 1 To Prop.Boundary.PropDischarges.NumberOfDischarges
    
      Ncell = Prop.Boundary.PropDischarges.Discharge(i).cell
      VolumeRate = Prop.Boundary.PropDischarges.Discharge(i).VolumeRate
      SpecificValue = Prop.Boundary.PropDischarges.Discharge(i).SpecificValue

End Sub
Sub ExpFreeSurfaceHeatFluxes(Temperature As Property, gr As grid, ctrl As Controls, atm As Atmosphere)
    
    Dim ro, cp, hfg, cB As Double
    Dim FluxConv As Double
    Dim FluxLatent As Double
    Dim FluxRadiation As Double
    Dim SolarFlux As Double
    Dim i As Integer
    
    ro = 1000 'kg/m3
    cp = 4180 'J/kg
    hfg = 2450000 'J/kg - Evaporation heat
    cB = 0.62 'mb/K
    
    
    For i = 1 To gr.NC
      SolarFlux = Solar_Radiation(ctrl.timedays, atm)
      Temperature.NewValue(i) = Temperature.NewValue(i) + ctrl.dt * SolarFlux / (ro * cp * gr.waterdepth)
    Next i

    
    If Temperature.Boundary.FreeSurfaceSensibleHeatFlux Then
    For i = 1 To gr.NC
      FluxConv = cB * h(atm.windspeed, atm.h_min) * (atm.Temperature - Temperature.NewValue(i)) 'W/m2
      Temperature.NewValue(i) = Temperature.NewValue(i) + ctrl.dt * FluxConv / (ro * cp * gr.waterdepth)
    Next i
    End If

    
    If Temperature.Boundary.FreeSurfaceLatentHeatFlux Then
    For i = 1 To gr.NC
      FluxLatent = -h(atm.windspeed, atm.h_min) * (Es(Temperature.NewValue(i)) - atm.Humidity * Es(atm.Temperature)) 'W/m2
       If (FluxLatent) > 0 Then
           FluxLatent = 0
       End If
      Temperature.NewValue(i) = Temperature.NewValue(i) + ctrl.dt * FluxLatent / (ro * cp * gr.waterdepth)
    Next i
    End If
    
    If Temperature.Boundary.FreeSurfaceRadiativeHeatFlux Then
    For i = 1 To gr.NC
      FluxRadiation = FluxRadiative(Temperature.NewValue(i), atm)
      Temperature.NewValue(i) = Temperature.NewValue(i) + ctrl.dt * FluxRadiation / (ro * cp * gr.waterdepth)
    Next i
    End If
   
End Sub
Function Solar_Radiation(timedays, atm As Atmosphere) As Double
Dim IM As Double
Dim Pi, AtmosphericAbsorption, hour As Double


   Pi = 3.14159265358979
   AtmosphericAbsorption = 0.23 ' Value measured by NASA
   
   IM = atm.SolarConstant * Cos(atm.Latitude * Pi / 180) * (1 - AtmosphericAbsorption)
   hour = (timedays - Fix(timedays)) * 24
   
   If hour < atm.Tsr Or hour > atm.Tss Then
   Solar_Radiation = 0
   Else
   Solar_Radiation = (1 - 0.75 * (atm.CloudCover) ^ 3) * IM * Sin(Pi * (hour - atm.Tsr) / (atm.Tss - atm.Tsr)) ' cloud cover solar radiation reduction according https://scool.larc.nasa.gov/lesson_plans/CloudCoverSolarRadiation.pdf
   End If
   
End Function

Function KGas(watertemperature As Double, velocity As Double, depth As Double)
Dim k As Double

k = 3.9 * ((Abs(velocity) + 0.001) / depth) ^ 0.5 ' the value 0.001 aims to guarantee that k is always larger than zero
KGas = k * 1.024 ^ (watertemperature - 20)

End Function

Function Kd_BOD(Oxygen As Double, BOD As Property, Temperature As Double) As Double
Dim kd As Double
    
   If Oxygen > 0 Then
      kd = BOD.Decayrate * (Oxygen / (BOD.GrazingKsat + Oxygen + 0.01))
      Kd_BOD = kd * 1.047 ^ (Temperature - 20)
   Else
      Kd_BOD = 0
   End If
    
End Function

Function getCsat_Henry(Prop As Property, WaterTemp As Double) As Double
 Dim J, JJ As Integer
 Dim Csat As Double
 Dim Slope, K_Henry_Interpolated As Double
  

 J = 1
 JJ = 1
Do While JJ <= Prop.Boundary.ValuesForFreeSurfaceGasExchange.N_HenryConstants
    JJ = JJ + 1
    If (WaterTemp > Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(1, J)) Then 'Temperature
          J = J + 1
    End If
Loop
 
 If J = 1 Then
  getCsat_Henry = Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(2, 1) * _
                  Prop.Boundary.ValuesForFreeSurfaceGasExchange.partialpressure * Prop.Boundary.ValuesForFreeSurfaceGasExchange.MolecularWeight 'Molecular weight is in mg/mole
 ElseIf J > Prop.Boundary.ValuesForFreeSurfaceGasExchange.N_HenryConstants Then
  getCsat_Henry = Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(2, Prop.Boundary.ValuesForFreeSurfaceGasExchange.N_HenryConstants) * _
                  Prop.Boundary.ValuesForFreeSurfaceGasExchange.partialpressure * Prop.Boundary.ValuesForFreeSurfaceGasExchange.MolecularWeight
 Else
   Slope = (Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(2, J) - Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(2, J - 1)) / _
           (Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(1, J) - Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(1, J - 1))
   K_Henry_Interpolated = Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(2, J - 1) + Slope * (WaterTemp - Prop.Boundary.ValuesForFreeSurfaceGasExchange.HenryConstValues(1, J - 1))
   getCsat_Henry = K_Henry_Interpolated * Prop.Boundary.ValuesForFreeSurfaceGasExchange.partialpressure * Prop.Boundary.ValuesForFreeSurfaceGasExchange.MolecularWeight

 End If

End Function
Function h(windspeed As Double, h_min As Double) As Double
   h = h_min + 0.345 * windspeed ^ 2
End Function
Function Es(T As Double) As Double

Const A = 6.112 'Pa
Const B = 17.67
Const C = 243.5 'ºC
Dim aux As Double

   aux = B * T / (T + C)
   Es = A * Exp(aux)
   
End Function
Function FluxRadiative(watertemperature As Double, atm As Atmosphere) As Double
Dim epson As Double
Dim Sigma As Double

epson = 0.97
Sigma = 5.67 * 10 ^ -8 'W/m2/k4
 If atm.SkyTemperatureImposed Then
    FluxRadiative = epson * Sigma * ((atm.SkyTemperature + 273.15) ^ 4 - (watertemperature + 273.15) ^ 4)
 Else
    FluxRadiative = epson * Sigma * ((9.37 * 10 ^ -6 * (atm.Temperature + 273.15) ^ 6 * (1 + 0.17 * atm.CloudCover ^ 2)) - (watertemperature + 273.15) ^ 4)
 End If
End Function

Sub ExpSinksBOD(pr As PropertiesList, gr As grid, dt As Double)
    
    Dim i As Integer
    Dim Decayrate, AerobicDecayRate, BODdecay, BODbeforeDecay, AnaerobicDecayRate, AnaerobicDecay, TemperatureAcceleration, aux As Double
    
    For i = 1 To gr.NC
    
    ' The logistic approach attempts to mimetize the natural growth of BOD due to primary production. Because BOD can be discharged
    ' from lateral boundaries it performs a test to the value of BOD. It is modified only if it is below the max value associated to the
    ' logistic equation. Otherwise the parenthesis woul become negative and it would generate an additional (unrealistic) decay
    ' It adds the decay to the growth rate because  phytoplankton growthrate is measured oindependently of the decay.
    ' The exponent 1/2 smoothes the curvature of the BOD profile when it is decaying from a peack of a discharge to the logistic value.
        
      If pr.BOD.NewValue(i) < pr.BOD.MaxValLogisticEquation Then
         pr.BOD.NewValue(i) = pr.BOD.NewValue(i) * (1 + dt * (pr.BOD.Decayrate + pr.BOD.GrowthRate) * (1 - pr.BOD.NewValue(i) / pr.BOD.MaxValLogisticEquation) ^ 0.5)
      End If
      
      BODbeforeDecay = pr.BOD.NewValue(i)
      TemperatureAcceleration = 1.047 ^ (pr.Temperature.NewValue(i) - 20)
      Decayrate = TemperatureAcceleration * pr.BOD.Decayrate * (pr.DO.NewValue(i) / (pr.BOD.GrazingKsat + pr.DO.NewValue(i) + 0.01)) ' adds 0.01 to minimize impacts of eventual negative values

      BODdecay = pr.BOD.NewValue(i) * Decayrate * dt
      
      If BODdecay > (BODbeforeDecay + pr.BOD.minimum) Then ' checks if available BOD is higher then decay
        BODdecay = BODbeforeDecay - pr.BOD.minimum
      End If
      
      If BODdecay > (pr.DO.NewValue(i) + pr.DO.minimum) Then ' checks if there is enough DO to mineralise the BODdecay
       BODdecay = pr.DO.NewValue(i) - pr.DO.minimum
      End If
      
      AerobicDecayRate = BODdecay / pr.BOD.NewValue(i) / dt 'value updated after the constrains
      
      pr.BOD.NewValue(i) = BODbeforeDecay - BODdecay        ' updates the values of the state variables
      pr.DO.NewValue(i) = pr.DO.NewValue(i) - BODdecay
      pr.CO2.NewValue(i) = pr.CO2.NewValue(i) + (BODdecay * (44 / 32)) 'stoichiometric ratio (CO2/O2)
      pr.BOD.IntegratedAerobicResp(i) = pr.BOD.IntegratedAerobicResp(i) + BODdecay * gr.dx * gr.RiverWidth * gr.waterdepth / 1000 ' to get kilograms per timestep
      
      If pr.BOD.AnaerobicRespiration Then  ' If DOD decay due to aerobic respiration is lower that the maximum, it adds the difference to anaerobic respiration
      
         AnaerobicDecayRate = TemperatureAcceleration * pr.BOD.Decayrate - AerobicDecayRate ' If anaerobic respiration is allowed, the total decayrate  will be the value imposed. The anaerobic respiration will not consume Oxygen
         aux = pr.BOD.NewValue(i)
         pr.BOD.NewValue(i) = pr.BOD.NewValue(i) / (1 + AnaerobicDecayRate * dt)  ' implicit calculation
         
         AnaerobicDecay = aux - pr.BOD.NewValue(i)

Sub ImpTypeSinksGeneric(Generic As Property, NC As Integer, ctrl As Controls)
        'Added by Ramiro
        
    Dim i As Integer
    
    'Old Saving the last computation
    'New will be computed
    For i = 1 To NC
        Generic.Old(i) = Generic.NewValue(i)
    Next i
    If ctrl.TimeDiscretisation = "imp" Then
       For i = 1 To NC
        Generic.NewValue(i) = Generic.Old(i) / (1 - Generic.Decayrate * ctrl.dt)
       Next i
    Else  ' else must be semi-implicit
       For i = 1 To NC
               Generic.NewValue(i) = Generic.NewValue(i) * (1 + Generic.Decayrate * ctrl.dt / 2) / (1 - Generic.Decayrate * ctrl.dt / 2)
       Next i
    End If
    
End Sub



Function minimum(ParamArray Values() As Variant)
'Determine Minimum Value from a Parameter Array (also works for nested array or a multiple column range)
'Returns minimum value from a Parameter Array (also works for nested array or a multiple column range).
   
Dim Item As Variant
Dim part As Variant
   
For Each Item In Values

  If IsArray(Item) Then
  'Checks if an item in the array of Values is itself an array (viz. nested array) and determines minimum value therein
      For Each part In Item
       minimum = minimum(part, minimum)
      Next
  Else
  'If an item in the array of Values is not an array
      If IsEmpty(minimum) Then
          minimum = Item
      Else
         If Item < minimum And Not IsEmpty(Item) Then
          minimum = Item
         End If
      End If

  End If
 
Next
   
End Function

     Do While (CELL_2IM1 < gr.NC) And Not (IsEmpty(ActiveSheet.Cells(LineNumber, Column)) Or IsEmpty(ActiveSheet.Cells(LineNumber, Column + 1)) Or IsEmpty(ActiveSheet.Cells(LineNumber + 1, Column)))
     
       interval = interval + 1

       CELL_1 = ActiveSheet.Cells(LineNumber, Column)
       CELL_2 = ActiveSheet.Cells(LineNumber, Column + 1)
       AuxValue = ActiveSheet.Cells(LineNumber + 1, Column)
       AuxCell = 0
            
       If (CELL_1 < 1 Or CELL_2 > gr.Length) Then
            MsgBox "property " & name & " Input error:  Invalid initialization interval values must be between 0 and" & gr.Length
            Stop
       ElseIf (CELL_1 < CELL_2IM1 Or CELL_2 < CELL_1) Then
            MsgBox "property " & name & " Input error: Intervals can't overlap"
            Stop
       Else
            AuxCell = CELL_1
            CreateProperty.NewValue(AuxCell) = AuxValue
            Do While (AuxCell <= CELL_2)
                CreateProperty.NewValue(AuxCell) = AuxValue
                AuxCell = AuxCell + 1
            Loop
            CELL_1IM1 = CELL_1
            CELL_2IM1 = CELL_2
       End If '(X1 < 0 Or X2 > gr.Length)
       
            Column = Column + 2
   Loop
   
       If (interval = 0) Then
             MsgBox "Input error on Property " & name & ". Invalid initialization per intervals: no interval found!!!, "
             Stop
       End If
   Else
             MsgBox "Input error on Property " & name & ". Invalid initialization per intervals: no keyword for interval values found!!!, "
             Stop
   
End If ' keyword = "(X1,X2)_intervalValues(m):"

End If 'InitType = "INTERVAL"
    
       

If (InitType = "CELL" Or InitType = "Cell" Or InitType = "cell") Then
     
     icounter = 0
     Success = False
     Do While keyword <> "CellNumber:" And icounter < 50
        icounter = icounter + 1
        keyword = ActiveSheet.Cells(icounter, 1)
        If keyword = "CellNumber:" Then
            LineNumber = icounter
            Success = True
        End If
      Loop

     If Success Then
       AuxCell = ActiveSheet.Cells(LineNumber, 2)
            
       If (AuxCell < 1 Or AuxCell > gr.NC) Then
            MsgBox "Input error: Invalid initialization cell value mut be between 1 and" & gr.NC & " in " & name
            End
       Else
            CreateProperty.NewValue(AuxCell) = ActiveSheet.Cells(LineNumber + 1, 2)
            Column = 3
       End If
     Else
            MsgBox "Input error: method CELL selected, but Keyword CellNumber: to read value not specified:  " & name
            End
     End If
     
     Do While (Not IsEmpty(ActiveSheet.Cells(LineNumber, Column)))
            AuxCell = ActiveSheet.Cells(LineNumber, Column)
          If (AuxCell < 1 Or AuxCell > gr.NC) Then
                MsgBox "Input error: Invalid initialization cell value mut be between 1 and" & gr.NC & name
                End
          Else
            CreateProperty.NewValue(AuxCell) = ActiveSheet.Cells(LineNumber + 1, Column)
            Column = Column + 1
          End If

      Loop
      
   End If 'InitType = "CELL"
   
     CreateProperty.Boundary.PropDischarges = CreatePropDischarges(name, gr)
    
End Function

Function CreateFlow(gr As grid, ctrl As Controls) As Flow_properties

    Dim i, icounter, LineNumber As Integer
    Dim AuxCell, Column As Integer
    Dim AuxValue As Double
    Dim keyword, auxtext As String
    Dim Success As Boolean

    Worksheets("River").Activate
 
 icounter = 0
 Success = False
  Do While keyword <> "FlowVelocity:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "FlowVelocity:" Then
            CreateFlow.velocity = ActiveSheet.Cells(icounter, 2)
            Success = True
    End If
  Loop

    If Not Success Then
            MsgBox "Input error: Velocity value for  must be specified in worksheet Flow"
            Stop
    End If
        
 icounter = 0
 Success = False
  Do While keyword <> "Diffusivity:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "Diffusivity:" Then
        auxtext = ActiveSheet.Cells(icounter, 2)
        If auxtext = "standard" Or auxtext = "Standard" Or auxtext = "std" Or auxtext = "STANDARD" Then
            CreateFlow.Diffusivity = 0.1 * Abs(CreateFlow.velocity) * gr.RiverWidth
            MsgBox "Diffusivity value to be used is " & CreateFlow.Diffusivity & " m2/s"
        Else
            CreateFlow.Diffusivity = ActiveSheet.Cells(icounter, 2)
        End If
        
            Success = True
    End If
  Loop

    If Not Success Then
            MsgBox "Input error: Diffusivity value for  must be specified in worksheet Flow"
            Stop
    End If
    
 icounter = 0
 Success = False
  Do While keyword <> "RiverSlope:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "RiverSlope:" Then
            CreateFlow.Riverslope = ActiveSheet.Cells(icounter, 2) / 100 ' reads as percentage
            Success = True
    End If
  Loop

    If Not Success Then
            MsgBox "Input error: RiverSlope value for  must be specified in worksheet Flow"
            Stop
    End If
    
    
    If ctrl.Advection Then
        CreateFlow.CourantNr = ctrl.dt * CreateFlow.velocity / (gr.dx)
    Else
        CreateFlow.CourantNr = 0#
    End If
    
    If ctrl.Diffusion Then
        CreateFlow.DiffusionNr = ctrl.dt * CreateFlow.Diffusivity / (gr.dx) ^ 2
    Else
        CreateFlow.DiffusionNr = 0#
    End If
     
    ActiveSheet.Cells(2, 6) = "Courant Number"
    ActiveSheet.Cells(2, 8) = Round(CreateFlow.CourantNr, 2)
    ActiveSheet.Cells(3, 6) = "Diffusion Number"
    ActiveSheet.Cells(3, 8) = Round(CreateFlow.DiffusionNr, 2)
    ActiveSheet.Cells(4, 6) = "Grid Reynolds Number"
    If CreateFlow.DiffusionNr > 0 Then
         ActiveSheet.Cells(4, 8) = Round(CreateFlow.CourantNr / CreateFlow.DiffusionNr, 2)
      Else
         ActiveSheet.Cells(4, 8) = "infinity"
    End If
    ActiveSheet.Cells(7, 6) = "Residence Time in the river (Days)"
    ActiveSheet.Cells(7, 8) = Round(gr.Length / (CreateFlow.velocity + 0.000001) / 3600 / 24, 1)
    ActiveSheet.Cells(8, 6) = "Residence Time in a Cell (Seconds)"
    ActiveSheet.Cells(8, 8) = Round(gr.dx / (CreateFlow.velocity + 0.000001), 2)
    
    
     
End Function

Function CreatePropDischarges(name As String, gr As grid) As DischargesList
   
   Dim ND, inputcolumn, num As Integer
   
    Dim i, icounter, LineNumber, callcounter As Integer
    Dim AuxCell, Column As Integer
    Dim AuxValue As Double
    Dim keyword, auxtext As String
    Dim Success As Boolean
    
   
    Worksheets("Discharges").Activate
    
 icounter = 0
 Success = False
  Do While keyword <> "DischargeNumbers:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "DischargeNumbers:" Then
            Success = True
    End If
  Loop
  
  If Not Success Then
  
       CreatePropDischarges.NumberOfDischarges = 0
       CreatePropDischarges.DischargesActive = False
       MsgBox "Warning: Keyword DischargeNumbers: not found in worksheet Discharge. Simulation will be carried without discharges"

       Exit Function

   Else 'counts the number of discharges and registers their ID numbers
        ND = 0
        num = 1
        inputcolumn = 2
      Do While (Not IsEmpty(ActiveSheet.Cells(icounter, inputcolumn)) And num > 0)
          num = ActiveSheet.Cells(icounter, inputcolumn)
          If (num > 0#) Then
              ND = ND + 1
              inputcolumn = inputcolumn + 1
          End If
       Loop
    
         CreatePropDischarges.NumberOfDischarges = ND
         If ND > 0 Then
             ReDim CreatePropDischarges.Discharge(ND)
         Else
              CreatePropDischarges.DischargesActive = False
              MsgBox "Keyword DischargeNumbers: found, but no discharge number found. Simulation will be carried without discharges"
              Exit Function
         End If
    End If 'success
     
 icounter = 0
 Success = False
  Do While keyword <> "DischargeNames:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "DischargeNames:" Then
            Success = True
    End If
  Loop
  
  If Success Then
           For i = 1 To ND
            CreatePropDischarges.Discharge(i).name = ActiveSheet.Cells(icounter, i + 1)
           Next i
  Else
            MsgBox "warning: Discharge names not found in worksheet Discharge. Simulation will continue"
  End If
  
  icounter = 0
  Success = False
  Do While keyword <> "DischargeCells:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "DischargeCells:" Then
            Success = True
    End If
  Loop
  
  If Success Then
           For i = 1 To ND
            CreatePropDischarges.Discharge(i).cell = ActiveSheet.Cells(icounter, i + 1)
               If CreatePropDischarges.Discharge(i).cell > gr.NC Then
                 MsgBox "warning: At least a Discharge is located at a Cell Number larger than " & gr.NC & " This and subsequent discharges will be ignored"
                 CreatePropDischarges.NumberOfDischarges = i - 1
                 Exit For
               End If
           Next i
  Else ' looks for keyword DischargeLocations:
  
     icounter = 0
     Do While keyword <> "DischargeLocations(km):" And icounter < 50
         icounter = icounter + 1
         keyword = ActiveSheet.Cells(icounter, 1)
         If keyword = "DischargeLocations(km):" Then
            Success = True
         End If
      Loop

      If Success Then
         For i = 1 To ND
           AuxValue = ActiveSheet.Cells(icounter, i + 1) * 1000 ' converts km into m
            CreatePropDischarges.Discharge(i).cell = AuxValue / gr.dx
               If CreatePropDischarges.Discharge(i).cell > gr.NC Then
                 MsgBox "warning: At least a Discharge is located off the domain. Length: " & gr.Length & " m. This and subsequent discharges will be ignored"
                 CreatePropDischarges.NumberOfDischarges = i - 1
                 Exit For
               End If
            Next i
          Else
            MsgBox "Input error: Keyword DischargeCells: or Keyword DischargeLocations(km): must be specified in worksheet Discharges"
            End
       End If 'DischargeLocations
    End If 'DischargeCells
  
  icounter = 0
  Success = False
  Do While keyword <> "DischargeFlowRates(m3/s):" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "DischargeFlowRates(m3/s):" Then
            Success = True
    End If
  Loop
  
  If Success Then
           For i = 1 To ND
            CreatePropDischarges.Discharge(i).VolumeRate = ActiveSheet.Cells(icounter, i + 1)
           Next i
  Else
            MsgBox "Input error: DischargeFlowRates(m3/s): not specified in worksheet Discharge"
            Stop
  End If
  
  If name = "Temperature" Then
             auxtext = "DischargeTemperatures:"
  ElseIf name = "DO" Then
             auxtext = "DischargeConcentrations_DO:"
  ElseIf name = "BOD" Then
             auxtext = "DischargeConcentrations_BOD:"
  ElseIf name = "CO2" Then
             auxtext = "DischargeConcentrations_CO2:"
  ElseIf name = "Generic" Then
             auxtext = "DischargeGeneric:"
  Else
            MsgBox "Input error: Variable " & name & " not known in worksheet Discharge"
            Stop
  End If
  
  icounter = 0
  Success = False
  Do While keyword <> auxtext And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = auxtext Then
            Success = True
    End If
  Loop
  
  If Success Then
  
           CreatePropDischarges.DischargesActive = True
           For i = 1 To ND
            CreatePropDischarges.Discharge(i).SpecificValue = ActiveSheet.Cells(icounter, i + 1)
            'If Name = "DO" Or Name = "BOD" Or Name = "CO2" Then 'It does not have to convert to mg/m3 because in "ApplyDischarges" the conversion cancels
              'CreatePropDischarges.Discharge(I).SpecificValue = CreatePropDischarges.Discharge(I).SpecificValue * 1000 ' converts to mg/L into mg/m3
            'End If
           Next i
  Else
           CreatePropDischarges.DischargesActive = False
            MsgBox "Warning: Specific values for property " & name & " not specified in worksheet Discharge. Discharges will not be applied"
            Exit Function

  End If
  

End Function

    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "NumberOfCells:" Then
        CreateGrid.NC = ActiveSheet.Cells(icounter, 2)
        Success = True
    End If
Loop

If (Not Success) Then
     MsgBox ("NumberOfCells: not found in  Worksheet Main")
     Stop
End If

' Reads Vertical area (cross section area)
icounter = 0
Success = False

Do While keyword <> "RiverWidth:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "RiverWidth:" Then
        CreateGrid.RiverWidth = ActiveSheet.Cells(icounter, 2)
        Success = True
    End If
Loop

If (Not Success) Then
     MsgBox ("RiverWidth: not found in  Worksheet Main")
     Stop
End If

' Reads water depth
icounter = 0
Success = False

Do While keyword <> "WaterDepth:" And icounter < 50
    icounter = icounter + 1
    keyword = ActiveSheet.Cells(icounter, 1)
    If keyword = "WaterDepth:" Then
        CreateGrid.waterdepth = ActiveSheet.Cells(icounter, 2)
        Success = True
    End If
Loop

If (Not Success) Then
     MsgBox ("WaterDepth: not found in  Worksheet Main")
     Stop
End If


' Computes dx and Horizontal Area
    
    CreateGrid.dx = CreateGrid.Length / CreateGrid.NC
    CreateGrid.AreaVertical = CreateGrid.waterdepth * CreateGrid.RiverWidth
    CreateGrid.AreaH = CreateGrid.RiverWidth * CreateGrid.dx
    
    ReDim CreateGrid.xc(CreateGrid.NC) As Double
       
' Computes cell center coordinate
    For i = 1 To CreateGrid.NC
        CreateGrid.xc(i) = CreateGrid.dx * (i - 0.5)
    Next
    
End Function

Sub CreateGraphs(pr As PropertiesList, ctrl As Controls, NC As Integer)

    Dim LastRow As Long, LastColumn As Long, LineToGraph As Long, PercentageSimulation As Long
    Dim NumOfTimeSeries As Integer, cell As Integer
    Dim i As Integer
    Dim NumberOfSpaceProfiles As Integer
    Dim NumberOfLines As Long
    Dim SeriesNumber As Integer
    Dim ws As Worksheet
    Dim RngX As Range, RngY As Range
    Dim PositionInSheetLeft As Double
    Dim PositionInSheetTop As Double
    Dim ChartObj As ChartObject
    
    
    Worksheets("Graph").Activate
    For Each ChartObj In Worksheets("Graph").ChartObjects
        ChartObj.Delete
    Next
        
'outside any numbers or formats
    Worksheets("Graph").Range("A10").Select
        
'Timeseries Type Graphs (Counts the number of cells were time series are performed)
    NumOfTimeSeries = Worksheets("Graph").Cells(2, Columns.Count).End(xlToLeft).Column - 2  'substracts 2 because cells start are provided starting at column 3
'Grid graphs (counts the number of longitudinal profiles desired)
    NumberOfSpaceProfiles = Worksheets("Graph").Cells(3, Columns.Count).End(xlToLeft).Column - 2 ' idem
    
    If NumOfTimeSeries > 0 Then
    
            For i = 1 To NumOfTimeSeries
            If (Worksheets("Graph").Cells(2, i + 2) < 0 Or Worksheets("Graph").Cells(2, i + 2) > NC) Then
                MsgBox "Cell number for graph is out of boundaries: " & Worksheets("Graph").Cells(2, i + 2)
                NumOfTimeSeries = i - 1
                Exit For
            End If
            Next i
            
        
        PositionInSheetLeft = 100
        PositionInSheetTop = 100
        
        Call CreateOverlapGraphsTime(pr, NumOfTimeSeries, ctrl, NC, PositionInSheetTop, PositionInSheetLeft) 'Plots all cells in the same graph

    End If
    
    
    
'space graphs
    If NumberOfSpaceProfiles > 0 Then
    
        For i = 1 To NumberOfSpaceProfiles
            If (Worksheets("Graph").Cells(3, i + 2) < 0 Or Worksheets("Graph").Cells(3, i + 2) > 100) Then
                MsgBox "Time Percentage for graph has to be between 0 and 100: " & Worksheets("Graph").Cells(3, i + 2)
                End
            End If
        Next i
        
            Call CreateOverlapGraphsSpace(pr, NumberOfSpaceProfiles, ctrl, NC, PositionInSheetTop, PositionInSheetLeft)

     End If
           

    If NumOfTimeSeries > 0 Then

        For i = 1 To NumOfTimeSeries ' Plots one graph per cell
               Call CreatePropertiesGraphsTime(pr, i, ctrl, PositionInSheetLeft, PositionInSheetTop)
            PositionInSheetTop = PositionInSheetTop + 350
            PositionInSheetLeft = 100
        Next
        
    End If
    
    If NumberOfSpaceProfiles > 0 Then
            
        For i = 1 To NumberOfSpaceProfiles

            NumberOfLines = ctrl.PrintLine - ctrl.FirstPrintLine - 1
            PercentageSimulation = CInt(Worksheets("Graph").Cells(3, i + 2))
  'what line to graph as a percentage of total lines
            LineToGraph = ctrl.FirstPrintLine + PercentageSimulation / 100 * NumberOfLines
               Call CreatePropertiesGraphsSpace(pr, LineToGraph, ctrl, PositionInSheetLeft, PositionInSheetTop, NC)
            PositionInSheetTop = PositionInSheetTop + 350
            PositionInSheetLeft = 100
        Next i
    
    End If
End Sub

         .HasTitle = True

         .Axes(xlCategory, xlPrimary).AxisTitle.Characters.Text = "Distance to origin (m)"
         .Axes(xlValue, xlPrimary).HasTitle = True
         .Axes(xlValue, xlPrimary).AxisTitle.Characters.Text = name

     End With

     Next i

          Call FormatPropertiesGraphs(PositionInSheetLeft, PositionInSheetTop, name, NumberOfSpaceProfiles)

Next
          PositionInSheetLeft = 100
          PositionInSheetTop = PositionInSheetTop + 350
